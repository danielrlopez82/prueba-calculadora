<input type="number" id="nicotineStrength" value="3">

<input type="number" id="baseNicotineStrength" value="20">

<input type="number" id="aromaPercent" value="10">

<label for="pgPercent">Proporción de PG deseado (%):</label>
<input type="number" id="pgPercent" value="50">

<label for="vgPercent">Proporción de VG deseado (%):</label>
<input type="number" id="vgPercent" value="50">

<button onclick="calculate()">Calcular</button>

<div class="results" id="results" style="display:none;">
  <h2>Resultados</h2>
  <p><strong>Aroma:</strong> <span id="aromaMl"></span> ml</p>
  <p><strong>Nicotina base:</strong> <span id="nicBaseMl"></span> ml</p>
  <p><strong>PG:</strong> <span id="pgMl"></span> ml</p>
  <p><strong>VG:</strong> <span id="vgMl"></span> ml</p>

<script>
  function calculate() {
    const totalMl = parseFloat(document.getElementById('totalMl').value);
    const nicotineStrength = parseFloat(document.getElementById('nicotineStrength').value);
    const baseNicotineStrength = parseFloat(document.getElementById('baseNicotineStrength').value);
    const aromaPercent = parseFloat(document.getElementById('aromaPercent').value);
    const pgPercent = parseFloat(document.getElementById('pgPercent').value);
    const vgPercent = parseFloat(document.getElementById('vgPercent').value);

    const aromaMl = totalMl * (aromaPercent / 100);
    const nicBaseMl = (totalMl * nicotineStrength) / baseNicotineStrength;
    const remainingMl = totalMl - aromaMl - nicBaseMl;

    const pgMl = remainingMl * (pgPercent / 100);
    const vgMl = remainingMl * (vgPercent / 100);

    document.getElementById('aromaMl').innerText = aromaMl.toFixed(2);
    document.getElementById('nicBaseMl').innerText = nicBaseMl.toFixed(2);
    document.getElementById('pgMl').innerText = pgMl.toFixed(2);
    document.getElementById('vgMl').innerText = vgMl.toFixed(2);
    document.getElementById('results').style.display = 'block';
  }
</script>
